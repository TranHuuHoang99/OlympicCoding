Kỹ thuật lazy update 

Mỗi node trên cây segment tree, ngoài giá trị maxValue, ta gọi thêm một giá trị lazyValue là tổng lượng update (nếu có yêu cầu tăng số trên đoạn [L ... R]) mà đang 

quản lý. 


struct itnode {
    int lazyValue; 
    int maxValue 
}


lazyUpdate(index, L, R):
  nếu it[index].lazyValue > 0 và L < R: 
    it[2 * index].lazyValue += it[index].lazyValue 
    it[2 * index].maxValue += it[index].lazyValue 
    it[2 * index + 1].lazyValue += it[index].lazyValue 
    it[2 * index + 1].maxValue += it[index].lazyValue 
  it[index].lazyValue = 0   


update(index, L, R, l, r, value): 
  lazyUpdate(index, L, R) 
  nếu l > R || L > r: 
    return 
  nếu l <= L && R <= r:  
    it[index].maxValue += value 
    it[index].lazyValue += value 
    lazyUpdate(index, L, R)
    return   
  mid = (L + R) / 2 
  update(2 * index, L, mid, l, r, value)
  update(2 * index + 1, mid + 1, R, l, r, value) 
  it[index].maxValue = max(it[2 * index].maxValue, it[2 * index + 1].maxValue) 


get(index, L, R, l, r)
  lazyUpdate(index, L, R) 
  nếu l > R || L > r:
    return -inf 
  nếu l <= L && R <= r: 
    return it[index].maxValue   
  mid = (L + R) / 2 
  đặt getLeft = get(2 * index, L, mid, l, r)
  đặt getRight = get(2 * index + 1, mid + 1, l, r) 
  return max(getLeft, getRight)
